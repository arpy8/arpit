#!/bin/bash ensure_in_path() { if [[ ":$PATH:" != *":$1:"* ]]; then export PATH="$1:$PATH" fi } if ! command -v python3 &> /dev/null then echo "Installing Python 3..." brew install python fi if [ -d "$HOME/.local/bin" ] ; then PATH="$HOME/.local/bin:$PATH" fi PYTHON_BIN_DIR="$(python3 -m site --user-base)/bin" ensure_in_path "$PYTHON_BIN_DIR" if ! command -v pip3 &> /dev/null then echo "Installing pip..." python3 -m ensurepip --upgrade fi echo "Installing bhxg package..." pip3 install --user git+https://gitlab.com/arpy8/bhxg if ! command -v bhxg &> /dev/null then echo "Error: bhxg was not installed correctly or is not in PATH." echo "Trying to find bhxg in Python bin directory..." if [ -f "$PYTHON_BIN_DIR/bhxg" ]; then echo "Found bhxg at $PYTHON_BIN_DIR/bhxg" chmod +x "$PYTHON_BIN_DIR/bhxg" ensure_in_path "$PYTHON_BIN_DIR" else echo "Could not find bhxg. Installation may have failed." exit 1 fi fi echo "Running bhxg flag..." bhxg flag || { echo "Error: Failed to run 'bhxg flag'. Trying with full path..." "$PYTHON_BIN_DIR/bhxg" flag }
