#!/bin/bash ensure_in_path() { if [[ ":$PATH:" != *":$1:"* ]]; then export PATH="$1:$PATH" fi } if ! command -v python3 &> /dev/null then echo "Installing Python 3..." brew install python fi if [ -d "$HOME/.local/bin" ] ; then PATH="$HOME/.local/bin:$PATH" fi PYTHON_BIN_DIR="$(python3 -m site --user-base)/bin" ensure_in_path "$PYTHON_BIN_DIR" if ! command -v pip3 &> /dev/null then echo "Installing pip..." python3 -m ensurepip --upgrade fi echo "Installing arpit package..." pip3 install --user git+https://gitlab.com/arpy8/arpit if ! command -v arpit &> /dev/null then echo "Error: arpit was not installed correctly or is not in PATH." echo "Trying to find arpit in Python bin directory..." if [ -f "$PYTHON_BIN_DIR/arpit" ]; then echo "Found arpit at $PYTHON_BIN_DIR/arpit" chmod +x "$PYTHON_BIN_DIR/arpit" ensure_in_path "$PYTHON_BIN_DIR" else echo "Could not find arpit. Installation may have failed." exit 1 fi fi echo "Running arpit..." arpit || { echo "Error: Failed to run 'arpit'. Trying with full path..." "$PYTHON_BIN_DIR/arpit" }
